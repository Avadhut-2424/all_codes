import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)

GPIO.setup(23, GPIO.OUT)
GPIO.setup(24, GPIO.IN)

try:
    while True:
        GPIO.output(23, 1)
        time.sleep(0.00001)
        GPIO.output(23, 0)

        while GPIO.input(24) == 0:
            start = time.time()

        while GPIO.input(24) == 1:
            end = time.time()

        print("Distance:", round((end - start) * 17150, 2), "cm")

        time.sleep(1)

except KeyboardInterrupt:
    GPIO.cleanup()

import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)

GPIO.setup(17, GPIO.IN)

while True:
    if GPIO.input(17) == 0:
        print("Object Detected")
    else:
        print("No Object")

    time.sleep(1)

3. buzzer+ relay


import RPi.GPIO as GPIO
import time

GPIO.setmode(GPIO.BCM)

GPIO.setup(27, GPIO.OUT)

try:
    while True:
        GPIO.output(27, GPIO.LOW)   # Relay ON
        print("ON")
        time.sleep(2)

        GPIO.output(27, GPIO.HIGH)  # Relay OFF
        print("OFF")
        time.sleep(2)

except KeyboardInterrupt:
    pass

finally:
    GPIO.cleanup()

4. server and client

#server____

import socket

s = socket.socket()
s.bind(("0.0.0.0", 5000))
s.listen(1)

conn, addr = s.accept()
print("Connected:", addr)

while True:
    data = conn.recv(1024).decode()

    if not data or data == "exit":
        break

    print("Client:", data)
    conn.send("Received".encode())

conn.close()


#client____

import socket

c = socket.socket()

c.connect(("192.168.1.100", 5000))  # Pi's IP

while True:
    msg = input("Enter: ")

    c.send(msg.encode())

    if msg == "exit":
        break

    print("Server:", c.recv(1024).decode())

c.close()
